Python Cheat Sheet - راهنمای سریع

1️⃣ Variables (متغیرها)

name = "Ali"
age = 20
height = 1.75
is_student = True

x, y = 1, 2

2️⃣ Print و String

print("Hello")

# اتصال رشته با +
print("Hello " + name)

# تبدیل عدد به رشته
print("Age: " + str(age))

# تکرار رشته با *
print("Hi " * 3)

# f-string (روش پیشنهادی)
print(f"{name} is {age} years old")

3️⃣ Input (گرفتن ورودی)

name = input("Enter name: ")

age = int(input("Enter age: "))
price = float(input("Enter price: "))

# input همیشه string برمی‌گرداند

4️⃣ If / Else

if age >= 18:
    print("Adult")
elif age >= 13:
    print("Teen")
else:
    print("Child")

5️⃣ Operators (عملگرها)

# ریاضی
+  -  *  /  %  //  **

# مقایسه‌ای
==  !=  >  <  >=  <=

# منطقی
and  or  not

# انتساب
=  +=  -=  *=  /=  %=

# مثال‌ها
x = 10
y = 3

print(x + y)    # 13 (جمع)
print(x - y)    # 7  (تفریق)
print(x * y)    # 30 (ضرب)
print(x / y)    # 3.33 (تقسیم اعشاری)
print(x // y)   # 3   (تقسیم صحیح)
print(x % y)    # 1   (باقیمانده)
print(x ** y)   # 1000 (توان)

x += 5         # x = x + 5
x *= 2         # x = x * 2

print(x > y and x < 20)  # True
print(not (x == y))       # True

6️⃣ For Loop و range

# از 0 تا قبل از 5
for i in range(5):
    print(i)

# start, stop
for i in range(1, 6):
    print(i)

# step
for i in range(0, 10, 2):
    print(i)

# معکوس با -1
for i in range(5, 0, -1):
    print(i)

for i in range(5, -1, -1):
    print(i)

# break برای خروج از حلقه
for i in range(10):
    if i == 3:
        break
    print(i)  # چاپ می‌کند: 0, 1, 2

# continue برای پرش از تکرار فعلی
for i in range(5):
    if i == 2:
        continue
    print(i)  # چاپ می‌کند: 0, 1, 3, 4

7️⃣ Lists (لیست)

numbers = [1, 2, 3]

# افزودن
numbers.append(4)
numbers.insert(1, 10)

# حذف
numbers.remove(2)
numbers.pop(0)

# دسترسی
print(numbers[0])

# تغییر مقدار
numbers[1] = 99

# اندازه لیست
print(len(numbers))

# اسلایس (بخشی از لیست)
print(numbers[1:4])  # از ایندکس 1 تا قبل از 4
print(numbers[:2])   # از ابتدا تا قبل از 2
print(numbers[2:])   # از ایندکس 2 تا انتها

# الحاق لیست‌ها
numbers.extend([5, 6])

# پیمایش
for n in numbers:
    print(n)

8️⃣ Dictionaries (دیکشنری)

person = {
    "name": "Ali",
    "age": 20
}

# افزودن یا بروزرسانی
person["city"] = "Tehran"
person["age"] = 21

# حذف
person.pop("age")
del person["city"]

# دسترسی
print(person["name"])

# دسترسی امن با get()
print(person.get("name", "Unknown"))
print(person.get("country", "Iran"))  # اگر وجود نداشت Iran برگردان

# بررسی وجود کلید
print("name" in person)
print("country" in person)

# اندازه دیکشنری
print(len(person))

# پیمایش کلیدها
for key in person:
    print(key)

# پیمایش کلید و مقدار
for key, value in person.items():
    print(key, value)
